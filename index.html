function parseGrid(grid, siteName) { let hRow = 2; for (let i = 0; i < Math.min(15, grid.length); i++) { if (grid[i].some(c => (c || '').toUpperCase().includes('CGM'))) { hRow = i; break; } } const above = hRow > 0 ? grid[hRow - 1] : []; const dayRow = grid[hRow]; const months = {}; let curMon = null; for (let c = 0; c < dayRow.length; c++) { const mv = (above[c] || '').trim(); const m1 = mv.match(/^(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[- ](\d{4})$/i); if (m1) { curMon = `${m1[1][0].toUpperCase() + m1[1].slice(1).toLowerCase()}-${m1[2]}`; } const dv = (dayRow[c] || '').trim(); const day = dv === '21st' ? 21 : parseInt(dv); if (!isNaN(day) && day >= 1 && day <= 31 && curMon) { if (!months[curMon]) months[curMon] = {}; months[curMon][day] = c; } } const recs = []; const PRIO = ['T', 'A', 'LM', 'LN', 'LO', 'LR']; let r = hRow + 1; while (r < grid.length) { const row = grid[r]; const cgmId = (row[0] || '').trim(); // ❌ skip non-driver rows if (!cgmId || !cgmId.startsWith('CGM')) { r++; continue; } // ✅ MERGED DATA (important fix) const driverName = (row[4] || '').trim(); const status = (row[3] || '').trim(); // ✅ dynamically collect merged rows (max 3) const block = []; let tempR = r; while ( tempR < grid.length && block.length < 3 && !(tempR !== r && (grid[tempR][0] || '').toUpperCase().startsWith('CGM')) ) { block.push(grid[tempR] || []); tempR++; } // ensure at least 3 rows while (block.length < 3) block.push([]); const [r1, r2, r3] = block; // 👉 COMBINE 3 ROWS DATA function getCell(c) { return (r1[c] || r2[c] || r3[c] || '').toString().trim(); } // 🔥 PROCESS MONTH DATA Object.keys(months).forEach(month => { const daysMap = months[month]; let total = 0; let dates = []; Object.keys(daysMap).forEach(d => { const col = daysMap[d]; const v1 = classify(r1[col]); const v2 = classify(r2[col]); const v3 = classify(r3[col]); const vals = [v1, v2, v3]; const found = PRIO.find(p => vals.includes(p)); if (found) { total++; dates.push(`${d}(${found})`); } }); if (total > 0) { const [monName, year] = month.split('-'); recs.push({ site: siteName, cgmId, driverName, status, month, year: parseInt(year), monNum: MON_MAP[monName], totalDays: total, leaveDates: dates.map(d => d.split('(')[0]).join(', '), datesWithCode: dates.join(', ') }); } }); // ✅ SMART JUMP (IMPORTANT) r = tempR; while (r < grid.length) { const nextVal = (grid[r][0] || '').trim().toUpperCase(); if (nextVal.startsWith('CGM')) break; r++; } } return recs; }
